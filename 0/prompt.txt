+
+After you add the component with the CLI, import it from your UI folder:
+
+```tsx
+"use client";
+
+import Counter from "@/components/prsui/counter";
+
+export const CounterPreview = () => (
+  <div className="flex flex-col items-center gap-8 py-12">
+    <p className="text-5xl font-medium tracking-tight text-foreground">

+
+The counter starts animating when it scrolls into view. Use `trigger="auto"` to start on mount instead.
+
+## Props
+
+| Prop               | Type                 | Default    | Description                                  |
+| ------------------ | -------------------- | ---------- | -------------------------------------------- |
+| `value`            | `number`             | -          | The final value to count up to.              |
+| `from`             | `number`             | `0`        | The value to start from.                     |
+| `duration`         | `number`             | `2`        | Animation duration in seconds.               |
+| `delay`            | `number`             | `0`        | Delay before the animation starts.           |
+| `decimals`         | `number`             | `0`        | Number of decimal places to display.         |
+| `prefix`           | `string`             | `""`       | Text rendered before the value.              |
+| `suffix`           | `string`             | `""`       | Text rendered after the value.               |
+| `separator`        | `string`             | `","`      | Thousands separator character.               |
+| `decimalSeparator` | `string`             | `"."`      | Decimal separator character.                 |
+| `transition`       | `Transition`         | -          | Overrides the animation transition.          |
+| `trigger`          | `"inView" \| "auto"` | `"inView"` | When to start the animation.                 |
+| `className`        | `string`             | -          | Classes applied to the counter element.      |
+| `onComplete`       | `() => void`         | -          | Callback fired when the animation completes. |

+  const inView = useInView(ref, { amount: 0.5, once: true });
+  const reduce = useReducedMotion();

+    }

+      delay,
+      duration: transition?.duration ?? duration,
+      ease: transition?.ease ?? EASE_OUT,
+      onComplete: () => {

+    });

+    duration,
+    delay,

+        initial="idle"

+        transition={{ duration: 0.3, ease: EASE_OUT }}
+      >
+        {prefix}
+        {formatValue(displayValue)}
+        {suffix}
+      </motion.span>
+    </span>
+  );
+};
